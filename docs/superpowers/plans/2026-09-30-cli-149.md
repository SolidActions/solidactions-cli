**Goal:** the `solidactions` CLI's skill and role commands work against today's server again:
- they call the current MCP tools
- they surface errors instead of printing `{}`
- they can pull a role and role-scoped skills, and publish a role-scoped skill

```ts
export type CrewsGroup = 'skills' | 'roles';

// Current catalog (solidactions-app ToolCatalog.php:61-100): the public tool depends on the action.
const CREWS_ROUTES: Record<CrewsGroup, Record<string, { tool: string; action?: string; rename?: Record<string, string> }>> = {
    skills: {
        list: { tool: 'crews_skills_read' },
        read: { tool: 'crews_skills_read' },
        search: { tool: 'crews_skills_read' },
        history: { tool: 'crews_skills_read' },
        read_reference_file: { tool: 'crews_skills_read' },
        create: { tool: 'crews_skills_manage' },
        edit: { tool: 'crews_skills_manage' },
        delete: { tool: 'crews_delete', action: 'delete_skill' },
        sandbox_exec: { tool: 'crews_sandbox', action: 'skill_exec', rename: { environment: 'skill_exec_environment' } },
    },
    roles: {
        list: { tool: 'crews_roles_read' },
        read: { tool: 'crews_roles_read' },
        list_skills: { tool: 'crews_roles_read' },
        read_skill: { tool: 'crews_roles_read' },
        read_file: { tool: 'crews_roles_read' },
        list_files: { tool: 'crews_roles_read' },
        read_reference_file: { tool: 'crews_roles_read' },
        create: { tool: 'crews_roles_manage' },
        edit: { tool: 'crews_roles_manage' },
        create_skill: { tool: 'crews_roles_manage' },
        edit_skill: { tool: 'crews_roles_manage' },
        sandbox_exec: { tool: 'crews_sandbox', action: 'role_exec', rename: { environment: 'role_exec_environment' } },
    },
};

export function resolveCrewsCall(group: CrewsGroup, args: Record<string, unknown>): { tool: string; args: Record<string, unknown> } {
    const action = String(args.action ?? '');
    const route = CREWS_ROUTES[group][action];
    if (!route) throw new Error(`unsupported crews action: ${group}.${action}`);
    const out: Record<string, unknown> = {};
    for (const [k, v] of Object.entries(args)) {
        if (v === undefined) continue;
        out[route.rename?.[k] ?? k] = v;
    }
    if (route.action) out.action = route.action;
    return { tool: route.tool, args: out };
}

function isGroup(name: string): name is CrewsGroup {
    return name === 'skills' || name === 'roles';
}

export async function callCrewsTool(config: Config, toolName: string, args: Record<string, unknown>): Promise<McpToolResult> {
    if (isGroup(toolName)) {
        const r = resolveCrewsCall(toolName, args);
        return callMcpTool(config, UNIFIED_MCP_PATH, r.tool, r.args);
    }
    return callMcpTool(config, UNIFIED_MCP_PATH, toolName, args);
}

export async function callCrewsToolContent(config: Config, toolName: string, args: Record<string, unknown>): Promise<McpContentResult> {
    const r = isGroup(toolName) ? resolveCrewsCall(toolName, args) : { tool: toolName, args };
    const raw = await postMcpTool(config, UNIFIED_MCP_PATH, r.tool, r.args);
    return { ok: !raw.isError, content: raw.content };
}
```

**Interfaces:**
- Produces:
  - `writeDirAtomic(dest: string, files: Record<string, string | Buffer>): void`. It writes every file into a fresh sibling temp dir (`${dest}.tmp-<pid>-<ts>`), then swaps it in: rename `dest` to `${dest}.old-<ts>` if it exists, rename temp to `dest`, and remove `.old`. On any error before the swap it removes the temp dir and rethrows, leaving `dest` untouched. Relative paths may contain `/` (subfolders are created). A path that is absolute or contains a `..` segment throws `Error("unsafe path: <p>")`.
  - `fetchSkillFiles(config: Config, name: string, opts?: { role?: string; inCrew?: string }): Promise<Record<string, string | Buffer>>`: reads, validates and returns the full file map (`SKILL.md`, `references/...`, sidecar) without writing anything
  - `pullSkillWithConfig(name, dest, config, opts?: { role?: string; inCrew?: string })`, which is `writeDirAtomic(dest, await fetchSkillFiles(...))`

```ts
import * as fs from 'fs';
import * as path from 'path';

export function writeDirAtomic(dest: string, files: Record<string, string | Buffer>): void {
    for (const rel of Object.keys(files)) {
        if (path.isAbsolute(rel) || rel.split(/[\\/]/).includes('..')) throw new Error(`unsafe path: ${rel}`);
    }
    const stamp = `${process.pid}-${Date.now()}`;
    const tmp = `${dest}.tmp-${stamp}`;
    fs.mkdirSync(tmp, { recursive: true });
    try {
        for (const [rel, content] of Object.entries(files)) {
            const target = path.join(tmp, rel);
            fs.mkdirSync(path.dirname(target), { recursive: true });
            fs.writeFileSync(target, content);
        }
    } catch (e) {
        fs.rmSync(tmp, { recursive: true, force: true });
        throw e;
    }
    const old = `${dest}.old-${stamp}`;
    const hadOld = fs.existsSync(dest);
    if (hadOld) fs.renameSync(dest, old);
    try {
        fs.renameSync(tmp, dest);
    } catch (e) {
        if (hadOld) fs.renameSync(old, dest);
        fs.rmSync(tmp, { recursive: true, force: true });
        throw e;
    }
    if (hadOld) fs.rmSync(old, { recursive: true, force: true });
}
```
